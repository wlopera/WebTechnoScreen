Para Dar permisos a poder subir y bajar archivos del storage de supabase:
--------------------------------------------------------------------------

-- 1. Permitir subir archivos (INSERT) al bucket 'pdfs'
CREATE POLICY "Permitir subidas públicas al bucket pdfs"
ON storage.objects FOR INSERT
TO public
WITH CHECK (bucket_id = 'pdfs');

-- 2. Permitir descargar y leer archivos (SELECT) del bucket 'pdfs'
CREATE POLICY "Permitir lecturas públicas al bucket pdfs"
ON storage.objects FOR SELECT
TO public
USING (bucket_id = 'pdfs');

-- 3. Permitir borrar archivos (DELETE) del bucket 'pdfs' (para que funcione el botón de eliminar de la consola Admin)
CREATE POLICY "Permitir eliminaciones públicas al bucket pdfs"
ON storage.objects FOR DELETE
TO public
USING (bucket_id = 'pdfs');
